     5→     5→import { motion, AnimatePresence } from "framer-motion";
     6→     6→import {

     8→     8→  Download, ChevronLeft, ChevronRight, ChevronsLeft, ChevronsRight,

    10→    10→} from "lucide-react";
    11→    11→import { Button } from "@/components/ui/button";
    12→    12→import { Input } from "@/components/ui/input";
    13→    13→import { Label } from "@/components/ui/label";
    14→    14→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    15→    15→import { Textarea } from "@/components/ui/textarea";
    16→    16→import { Card } from "@/components/ui/card";
    17→    17→import {
    18→    18→  Dialog, DialogContent, DialogHeader, DialogTitle
    19→    19→} from "@/components/ui/dialog";
    20→    20→import { apiGet, apiPost, apiPatch, apiDelete, formatEGP, formatDateTime, STATUS_LABELS, STATUS_COLORS } from "@/lib/api";
    21→    21→import { toast } from "sonner";
    22→    22→
    23→    23→const STATUSES = ["pending", "picked_up", "in_transit", "out_for_delivery", "delivered", "returned", "cancelled"];
    24→    24→
    25→    25→export function DashboardShipments() {
    26→    26→  const qc = useQueryClient();
    27→    27→  const [q, setQ] = useState("");
    28→    28→  const [status, setStatus] = useState("");
    29→    29→  const [page, setPage] = useState(1);
    30→    30→  const [pageSize, setPageSize] = useState(8);
    31→    31→  const [modalOpen, setModalOpen] = useState(false);
    32→    32→  const [drawerShipment, setDrawerShipment] = useState<any>(null);
    33→    33→  const [selected, setSelected] = useState<Set<string>>(new Set());
    34→    34→  const [editing, setEditing] = useState<any>(null);
    35→    35→  const [saving, setSaving] = useState(false);
    36→    36→  const [form, setForm] = useState<any>({
    37→    37→    senderName: "", senderPhone: "", senderAddress: "",
    38→    38→    receiverName: "", receiverPhone: "", receiverAddress: "",
    39→    39→    governorate: "", service: "Standard", weight: 1, notes: "", price: 0, status: "pending",
    40→    40→  });

    42→    42→  const { data: govData } = useQuery({ queryKey: ["governorates"], queryFn: () => apiGet<any>("/api/governorates") });
    43→    43→  const { data: svcData } = useQuery({ queryKey: ["services"], queryFn: () => apiGet<any>("/api/services") });

    45→    45→    queryKey: ["shipments", q, status],
    46→    46→    queryFn: () => apiGet<any>(`/api/shipments?q=${encodeURIComponent(q)}&status=${encodeURIComponent(status)}`),
    47→    47→  });

    49→    49→

    55→    55→    return allShipments.slice(start, start + pageSize);
    56→    56→  }, [allShipments, currentPage, pageSize]);
    57→    57→

    59→    59→
    60→    60→  const toggleSelect = (id: string) => {

    62→    62→    if (next.has(id)) next.delete(id); else next.add(id);
    63→    63→    setSelected(next);
    64→    64→  };

    66→    66→    const next = new Set(selected);
    67→    67→    if (allSelected) pagedShipments.forEach((s: any) => next.delete(s.id));
    68→    68→    else pagedShipments.forEach((s: any) => next.add(s.id));
    69→    69→    setSelected(next);
    70→    70→  };
    71→    71→
    72→    72→  const openCreate = () => {
    73→    73→    setEditing(null);
    74→    74→    setForm({ senderName: "Redline Warehouse", senderPhone: "019000", senderAddress: "5th Settlement, New Cairo",
    75→    75→      receiverName: "", receiverPhone: "", receiverAddress: "", governorate: govData?.governorates?.[0]?.name || "Cairo",
    76→    76→      service: "Standard", weight: 1, notes: "", price: 0, status: "pending" });
    77→    77→    setModalOpen(true);
    78→    78→  };
    79→    79→
    80→    80→  const openEdit = (sh: any) => {
    81→    81→    setEditing(sh);
    82→    82→    setForm({ ...sh });
    83→    83→    setModalOpen(true);
    84→    84→  };

    86→    86→  const save = async () => {
    87→    87→    setSaving(true);

    89→    89→      if (editing) {
    90→    90→        await apiPatch(`/api/shipments/${editing.id}`, form);
    91→    91→        toast.success("تم تحديث الشحنة");
    92→    92→      } else {
    93→    93→        await apiPost("/api/shipments", form);
    94→    94→        toast.success("تم إنشاء الشحنة");
    95→    95→      }
    96→    96→      setModalOpen(false);
    97→    97→      qc.invalidateQueries({ queryKey: ["shipments"] });
    98→    98→      qc.invalidateQueries({ queryKey: ["dashboard-stats"] });
    99→    99→    } catch (e: any) {
   100→   100→      toast.error(e.message || "خطأ");
   101→   101→    } finally {
   102→   102→      setSaving(false);
   103→   103→    }
   104→   104→  };
   105→   105→
   106→   106→  const remove = async (id: string) => {
   107→   107→    if (!confirm("هل أنت متأكد من حذف هذه الشحنة؟")) return;
   108→   108→    try {
   109→   109→      await apiDelete(`/api/shipments/${id}`);
   110→   110→      toast.success("تم الحذف");
   111→   111→      qc.invalidateQueries({ queryKey: ["shipments"] });
   112→   112→      qc.invalidateQueries({ queryKey: ["dashboard-stats"] });
   113→   113→    } catch (e: any) {
   114→   114→      toast.error(e.message);
   115→   115→    }
   116→   116→  };
   117→   117→
   118→   118→  const bulkDelete = async () => {
   119→   119→    if (selected.size === 0) return;
   120→   120→    if (!confirm(`حذف ${selected.size} شحنة؟`)) return;
   121→   121→    let ok = 0;
   122→   122→    for (const id of selected) {
   123→   123→      try { await apiDelete(`/api/shipments/${id}`); ok++; } catch {}
   124→   124→    }
   125→   125→    toast.success(`تم حذف ${ok} شحنة`);
   126→   126→    setSelected(new Set());
   127→   127→    qc.invalidateQueries({ queryKey: ["shipments"] });
   128→   128→    qc.invalidateQueries({ queryKey: ["dashboard-stats"] });
   129→   129→  };
   130→   130→
   131→   131→  const quickStatus = async (sh: any, newStatus: string) => {

   133→   133→      await apiPatch(`/api/shipments/${sh.id}`, { status: newStatus, note: "تحديث سريع من اللوحة" });
   134→   134→      toast.success("تم تحديث الحالة");
   135→   135→      qc.invalidateQueries({ queryKey: ["shipments"] });
   136→   136→      qc.invalidateQueries({ queryKey: ["dashboard-stats"] });
   137→   137→    } catch (e: any) {
   138→   138→      toast.error(e.message);
   139→   139→    }
   140→   140→  };
   141→   141→
   142→   142→  const exportCSV = () => {

   147→   147→    const url = URL.createObjectURL(blob);
   148→   148→    const a = document.createElement("a");
   149→   149→    a.href = url;
   150→   150→    a.download = `redline-shipments-${Date.now()}.csv`;
   151→   151→    a.click();
   152→   152→    URL.revokeObjectURL(url);
   153→   153→    toast.success("تم تصدير الشحنات إلى CSV");
   154→   154→  };
   155→   155→
   156→   156→  return (

   187→   187→            >
   188→   188→              <div className="mt-3 flex items-center justify-between p-3 rounded-xl bg-primary/5 border border-primary/20">

   191→   191→                  <Button variant="outline" size="sm" className="h-8" onClick={() => setSelected(new Set())}>إلغاء</Button>
   192→   192→                  <Button variant="outline" size="sm" className="h-8 text-rose-600" onClick={bulkDelete}>
   193→   193→                    <Trash2 className="w-3.5 h-3.5" /> حذف المحدد
   194→   194→                  </Button>
   195→   195→                </div>
   196→   196→              </div>
   197→   197→            </motion.div>
   198→   198→          )}

   200→   200→      </Card>

   202→   202→      {/* Table */}

   205→   205→          <div className="flex justify-center py-16"><Loader2 className="w-6 h-6 animate-spin text-primary" /></div>
   206→   206→        ) : pagedShipments.length === 0 ? (
   207→   207→          <div className="flex flex-col items-center py-16 text-muted-foreground">
   208→   208→            <Package className="w-12 h-12 mb-3 opacity-40" />
   209→   209→            <p>لا توجد شحنات مطابقة</p>
   210→   210→          </div>
   211→   211→        ) : (
   212→   212→          <>
   213→   213→            <div className="overflow-x-auto">
   214→   214→              <table className="w-full text-sm">
   215→   215→                <thead className="bg-muted/60">
   216→   216→                  <tr className="text-right">
   217→   217→                    <th className="px-4 py-3 w-10">
   218→   218→                      <button onClick={toggleAll} className="flex items-center">
   219→   219→                        {allSelected ? <CheckSquare className="w-4 h-4 text-primary" /> : <Square className="w-4 h-4 text-muted-foreground" />}
   220→   220→                      </button>
   221→   221→                    </th>
   222→   222→                    <th className="px-4 py-3 font-semibold">رقم التتبع</th>
   223→   223→                    <th className="px-4 py-3 font-semibold">المستلم</th>
   224→   224→                    <th className="px-4 py-3 font-semibold hidden md:table-cell">المحافظة</th>

   231→   231→                <tbody>

   233→   233→                    <tr key={sh.id} className={`border-t border-border hover:bg-muted/30 transition-colors ${selected.has(sh.id) ? "bg-primary/5" : ""}`}>
   234→   234→                      <td className="px-4 py-3">
   235→   235→                        <button onClick={() => toggleSelect(sh.id)}>
   236→   236→                          {selected.has(sh.id) ? <CheckSquare className="w-4 h-4 text-primary" /> : <Square className="w-4 h-4 text-muted-foreground" />}
   237→   237→                        </button>
   238→   238→                      </td>
   239→   239→                      <td className="px-4 py-3">
   240→   240→                        <button onClick={() => setDrawerShipment(sh)} className="font-mono font-bold text-primary hover:underline">
   241→   241→                          {sh.trackingNo}

   243→   243→                      </td>
   244→   244→                      <td className="px-4 py-3">
   245→   245→                        <div className="font-medium">{sh.receiverName}</div>
   246→   246→                        <div className="text-xs text-muted-foreground" dir="ltr">{sh.receiverPhone}</div>
   247→   247→                      </td>
   248→   248→                      <td className="px-4 py-3 hidden md:table-cell">{sh.governorate}</td>
   249→   249→                      <td className="px-4 py-3 hidden lg:table-cell">{sh.service}</td>
   250→   250→                      <td className="px-4 py-3 font-semibold tabular-nums">{formatEGP(sh.price)}</td>
   251→   251→                      <td className="px-4 py-3">
   252→   252→                        <Select defaultValue={sh.status} onValueChange={(v) => quickStatus(sh, v)}>
   253→   253→                          <SelectTrigger className="h-8 w-36 text-xs border-0 hover:bg-muted">
   254→   254→                            <SelectValue>
   255→   255→                              <span className={`px-2 py-0.5 rounded-full text-[10px] font-bold border ${STATUS_COLORS[sh.status] || ""}`}>
   256→   256→                                {STATUS_LABELS[sh.status] || sh.status}
   257→   257→                              </span>
   258→   258→                            </SelectValue>
   259→   259→                          </SelectTrigger>
   260→   260→                          <SelectContent>
   261→   261→                            {STATUSES.map((s) => <SelectItem key={s} value={s}>{STATUS_LABELS[s]}</SelectItem>)}
   262→   262→                          </SelectContent>
   263→   263→                        </Select>
   264→   264→                      </td>
   265→   265→                      <td className="px-4 py-3">
   266→   266→                        <div className="flex gap-1 justify-end">
   267→   267→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setDrawerShipment(sh)} title="عرض">
   268→   268→                            <Search className="w-4 h-4" />
   269→   269→                          </Button>
   270→   270→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openEdit(sh)}>
   271→   271→                            <Pencil className="w-4 h-4" />
   272→   272→                          </Button>
   273→   273→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-rose-600 hover:text-rose-700" onClick={() => remove(sh.id)}>
   274→   274→                            <Trash2 className="w-4 h-4" />
   275→   275→                          </Button>
   276→   276→                        </div>
   277→   277→                      </td>
   278→   278→                    </tr>
   279→   279→                  ))}
   280→   280→                </tbody>
   281→   281→              </table>
   282→   282→            </div>
   283→   283→
   284→   284→            {/* Pagination */}
   285→   285→            <div className="flex flex-col sm:flex-row items-center justify-between gap-3 p-4 border-t border-border">
   286→   286→              <div className="flex items-center gap-2 text-sm text-muted-foreground">
   287→   287→                <span>عرض</span>
   288→   288→                <Select value={String(pageSize)} onValueChange={(v) => { setPageSize(Number(v)); setPage(1); }}>
   289→   289→                  <SelectTrigger className="h-8 w-20"><SelectValue /></SelectTrigger>
   290→   290→                  <SelectContent>
   291→   291→                    {[8, 12, 20, 50].map((n) => <SelectItem key={n} value={String(n)}>{n}</SelectItem>)}
   292→   292→                  </SelectContent>
   293→   293→                </Select>
   294→   294→                <span>من {allShipments.length} شحنة</span>
   295→   295→              </div>
   296→   296→              <div className="flex items-center gap-1">

   305→   305→                  <ChevronLeft className="w-4 h-4" />
   306→   306→                </Button>
   307→   307→                <Button variant="outline" size="icon" className="h-8 w-8" disabled={currentPage === totalPages} onClick={() => setPage(totalPages)}>
   308→   308→                  <ChevronsLeft className="w-4 h-4" />
   309→   309→                </Button>
   310→   310→              </div>
   311→   311→            </div>
   312→   312→          </>
   313→   313→        )}
   314→   314→      </Card>
   315→   315→
   316→   316→      {/* Create/Edit Modal */}
   317→   317→      <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   318→   318→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   319→   319→          <DialogHeader>
   320→   320→            <DialogTitle className="flex items-center gap-2">
   321→   321→              <Package className="w-5 h-5 text-primary" />
   322→   322→              {editing ? "تعديل الشحنة" : "شحنة جديدة"}

   324→   324→          </DialogHeader>
   325→   325→          <div className="space-y-4">
   326→   326→            <div className="grid sm:grid-cols-2 gap-3">

   341→   341→            <div className="grid sm:grid-cols-2 gap-3">
