  1009→interface I18nState {
  1010→  locale: Locale;
  1011→  setLocale: (l: Locale) => void;
  1012→}
  1013→
  1014→export const useI18n = create<I18nState>((set) => ({
  1015→  locale: "ar",
  1016→  setLocale: (locale) => {
  1017→    set({ locale });
  1018→    if (typeof document !== "undefined") {
  1019→      document.documentElement.lang = locale;
  1020→      document.documentElement.dir = locale === "ar" ? "rtl" : "ltr";
  1021→    }
  1022→    try { localStorage.setItem("redline-locale", locale); } catch {}
  1023→  },
  1024→}));
  1025→
  1026→// === Hook: useT (returns t function + current locale) ===
  1027→export function useT() {
  1028→  const locale = useI18n((s) => s.locale);
  1029→  const t = (key: string): string => MESSAGES[locale]?.[key] ?? MESSAGES.ar[key] ?? key;
  1030→  return { t, locale, isAr: locale === "ar", isEn: locale === "en" };
  1031→}
  1032→
  1033→// === Init locale from storage ===
  1034→if (typeof window !== "undefined") {
  1035→  try {
  1036→    const saved = localStorage.getItem("redline-locale") as Locale | null;
  1037→    if (saved) {
  1038→      useI18n.getState().setLocale(saved);
  1039→    }
  1040→  } catch {}
  1041→}
